# v1.2.2 M3 · Word/Excel 在线模板编辑器扩展 + 文书生成打印 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有 Excel 模板设计器（`form-excel-designer.tsx` + `form-excel-config.ts`）基础上扩展：① `DocumentTemplate.type` 增加 `WORD`；② 新增 Word 设计器：A4 画布 + 所见即所得 + 字段库（调查/征收/系统 + 公式/条件显示）；③ 后端渲染：Excel 用现有 ExcelJS，Word 改用 `docx`（= docx.js，npm）生成 `.docx`；④ 生成&打印：Dashboard/H5 列表/详情页的「生成文书」按钮 → 选模板 → 预览 → 下载/打印；⑤ 所有生成/下载/打印全部写 `DocumentGenLog`（M1 已加表）。

**Architecture:** 前端设计器两种模式复用同一套字段库（`fieldLibrary.ts` 提取公共字段），两种画布分开：Excel 画布沿用现在 form-excel-designer（不删不改），Word 画布走 `word-designer.tsx`（contentEditable A4 预览 + 拖拽字段插入 `{{fieldX}}` 占位符）；后端渲染统一 route：`/api/document-templates/[id]/render?recordId=xxx` → 按 type 返回 `{downloadUrl, blobName}`；Excel 渲染仍走现有 ExcelJS 逻辑，Word 渲染走新的 `renderWordDocx(template.config, data)` 用 `docx` packer 生成 Blob。

**Tech Stack:** 前端现有 Tiptap/prosemirror？若无就直接用 `contenteditable`（避免额外重型 RTE，对 A4 简单文档完全够用）。后端新依赖 `docx`（npm，纯 TypeScript，无系统依赖），`file-saver` 可复用（已有）。

// web/lib/word-renderer-server.ts
import { Packer, Document, Paragraph, TextRun, AlignmentType, Table, TableRow, TableCell, WidthType, PageBreak, Footer, Header } from 'docx';
import type { WordDocumentConfig, WordBlock } from '@/types/document-template-shared';

// —— 工具：替换 {{fieldId}} → data[fieldId]；支持 toRMB / 日期格式化 ——
function evalString(s: string, data: any) {
  return s.replace(/\{\{([a-zA-Z0-9_:\.-]+)\}\}/g, (_, k) => {
    if (k.startsWith('system:')) return data[k] ?? '';
    if (k.startsWith('formula:rmb_upper')) return ''; // 实现略
    return data[k] ?? '';
  });
}

function blockToParagraph(b: WordBlock, data: any): Paragraph | null {
  if (b.type === 'PAGE_BREAK') return new Paragraph({ children: [new PageBreak() as any] });
  if (b.type === 'CONDITION_WRAPPER') {
    const ok = evalCondition(b.conditionExpr!, data);
    if (!ok) return null;
    // 渲染内部 children（每个 child 各自 paragraph）
  }
  const style = b.style ?? {};
  const text = evalString(b.content ?? '', data);
  return new Paragraph({
    alignment: (style.align === 'center' ? AlignmentType.CENTER : style.align === 'right' ? AlignmentType.RIGHT : AlignmentType.JUSTIFIED) as any,
    spacing: { before: style.marginBottomMm ? style.marginBottomMm*56 : undefined,
               after:  style.marginTopMm    ? style.marginTopMm*56    : undefined },
    indent: style.indentFirstLineChars ? { firstLine: (style.fontSizePt ?? 12) * style.indentFirstLineChars * 2 } : undefined,
    children: [new TextRun({
      text,
      bold: style.bold,
      italics: style.italic,
      underline: style.underline ? {} : undefined,
      size: style.fontSizePt ? style.fontSizePt * 2 : undefined, // docx half-points
      font: style.fontFamily?.includes('SimSun') ? '宋体' : (style.fontFamily?.includes('SimHei') ? '黑体' : undefined),
      color: style.colorHex?.replace('#', ''),
    })],
  });
}
function evalCondition(expr: string, data: any): boolean {
  try { return Function('data', 'Math', `with(data){ return !!(${expr}); }`)(data, Math); }
  catch { return false; }
}
function tableToDocx(rows: NonNullable<WordBlock['rows']>, data: any): Table {
  return new Table({ width: { size: 100, type: WidthType.PERCENTAGE }, rows: rows.map(r =>
    new TableRow({ children: r.map(cell => new TableCell({ width: cell.widthMm ? { size: cell.widthMm*56, type: WidthType.DXA } : undefined,
      children: (cell.blocks.map(b => blockToParagraph(b, data)).filter(Boolean) as Paragraph[]) }))
  )});
}

export async function renderDocxToBuffer(cfg: WordDocumentConfig, data: any): Promise<Buffer> {
  const sectionsChunks: Paragraph[] = [];
  for (const b of cfg.sections) {
    if (b.type === 'TABLE') { sectionsChunks.push(tableToDocx(b.rows!, data) as any); continue; }
    const p = blockToParagraph(b, data);
    if (p) sectionsChunks.push(p);
  }
  const doc = new Document({
    styles: { default: { document: { run: { size: (cfg.defaultStyle.fontSizePt ?? 12)*2 } } } },
    sections: [{
      properties: {},
      headers: cfg.header?.length ? { default: new Header({ children: cfg.header.map(b=>blockToParagraph(b,data)!).filter(Boolean) }) } : undefined,
      footers: cfg.footer?.length ? { default: new Footer({ children: cfg.footer.map(b=>blockToParagraph(b,data)!).filter(Boolean) }) } : undefined,
      children: sectionsChunks,
    }],
  });
  return Buffer.from(await Packer.toArray(doc));
}

// web/app/api/document-templates/[id]/render/route.ts
import { NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { renderDocxToBuffer } from '@/lib/word-renderer-server';
export async function GET(req: Request, { params }: { params: { id: string }}) {
  const url = new URL(req.url);
  const recordId = url.searchParams.get('recordId');
  const tpl = await prisma.documentTemplate.findUnique({ where: { id: params.id }});
  if (!tpl) return NextResponse.json({ code: 404 }, { status: 404 });
  // 1. 取 record 数据（含关联 LEVY_RELATION 调查记录展开）
  const data = await collectRecordDataForRender(tpl.tableIds as string[] ?? [], recordId!);
  let buf: Buffer;
  let filename: string;
  if (tpl.type === 'WORD') {
    buf = await renderDocxToBuffer(tpl.config as any, data);
    filename = `${tpl.name}_${new Date().toISOString().slice(0,10)}.docx`;
  } else {
    // EXCEL：复用现有 excel 渲染（从 form-excel-config 对应 route 迁移）
    buf = Buffer.alloc(0); // TODO: 这里放现有 excel render 调用
    filename = `${tpl.name}.xlsx`;
  }
  // 2. 写 DocumentGenLog（M1 已加）
  await prisma.documentGenLog.create({ data: {
    templateId: tpl.id, tableIds: tpl.tableIds, recordId: recordId!,
    action: 'DOWNLOAD', operatorId: (await currentUser())?.id ?? null, fileSizeBytes: buf.length,
    fileType: tpl.type,
  } as any });
  // 3. 返回下载
  return new NextResponse(buf, { status:200,
    headers: { 'Content-Type': tpl.type==='WORD'
      ? 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
      : 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
      'Content-Disposition': `attachment; filename="${encodeURIComponent(filename)}"` }});
}
