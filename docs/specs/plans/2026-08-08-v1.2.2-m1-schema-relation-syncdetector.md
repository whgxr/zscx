> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 v1.2.2 征收模块奠定数据库 / 枚举 / 字段类型 / 同步检测能力的基础骨架，跑通 Prisma 迁移 + 种子 + 新字段 LEVY_RELATION 在表单设计器可配且保存时校验；调查侧保存后自动为关联征收记录生成 DataSyncRequest。

**Tech Stack:** Prisma ORM + PostgreSQL, Next.js App Router (Route Handlers `web/app/api/*`), React + TypeScript (`web/components`, `web/app/dashboard`), Tailwind CSS + shadcn/ui 复用已有组件。新依赖：暂 0（docx.js / pdf.js 放 M3；Redis/BullMQ 暂不引入用 Promise 异步）。

```ts
if (body.fieldType === 'LEVY_RELATION') {

  if (!cfg?.surveyTableId || !cfg?.displayFieldId || !Array.isArray(cfg.fieldMapping)) {

  }
  // fieldMapping 里的 surveyFieldId / levyFieldId 都要真实存在
  const allSurveyFieldIds = new Set((await prisma.tableField.findMany({ where: { tableId: cfg.surveyTableId }})).map(f=>f.id));
```

```ts
// web/lib/levy-sync-detector.ts
import { prisma } from './prisma';
import type { Table, CategoryModule } from '@prisma/client';

/**
 * 调查侧保存成功后触发：查出所有 LEVY_RELATION → 关联到本调查 recordId 的征收记录
 * 计算 fieldMapping 差异 -> 存 SYNC_BEFORE / SYNC_AFTER 快照 -> 生成 PENDING DataSyncRequest
 * 如果同一 levyRecordId 已经存在 status=PENDING/IN_APPROVAL 的记录，则合并覆盖
 */
export async function triggerSyncForSurveyRecordIfNeeded(
  surveyTableId: string, surveyRecordId: string, newSurveyData: any,
  operatorId: string | null = null,
): Promise<void> {
  // 1. 查引用这张调查表且 autoSyncOnSurveyChange=true 的所有 LEVY_RELATION 字段
  const relationFields = await prisma.$queryRawUnsafe<any[]>(`
    SELECT f.id, f."tableId" as "levyTableId", f.config
    FROM "TableField" f
    JOIN "Table" t ON t.id = f."tableId"
    JOIN "TableCategory" c ON c.id = t."categoryId"
    WHERE f."fieldType" = 'LEVY_RELATION'::"FieldType"
      AND c.module = 'LEVY'::"CategoryModule"
      AND (f.config->'levyRelationConfig'->>'surveyTableId') = $1
      AND (f.config->'levyRelationConfig'->>'autoSyncOnSurveyChange')::boolean = true
  `, surveyTableId);

  for (const rf of relationFields) {
    const cfg = rf.config?.levyRelationConfig;
    if (!cfg) continue;
    // 2. 查 levyTableId 下，该 LEVY_RELATION 字段 value == surveyRecordId 的 DataRecord（已存在的征收记录）
    const colName = `"data"`; // Postgres JSONB
    const relatedLevyRecords = await prisma.$queryRawUnsafe<any[]>(`
      SELECT id, data FROM "DataRecord"
       WHERE "tableId" = $1
         AND data->>${rf.id::text} = $2
         AND "deletedAt" IS NULL
    `, rf.levyTableId, surveyRecordId);

    for (const rec of relatedLevyRecords) {
      const { beforeSnapshot, afterSnapshot, changedFields } = await buildSyncSnapshots(
        rf.levyTableId, rec, cfg.fieldMapping, newSurveyData, rf.id,
      );
      if (Object.keys(changedFields).length === 0) continue; // 无差异跳过

      // 3. 合并策略：同一 levyRecordId + levyRelationFieldId 有 PENDING/IN_APPROVAL，直接覆盖 diff
      const existing = await prisma.dataSyncRequest.findFirst({
        where: { levyRecordId: rec.id, levyRelationFieldId: rf.id, status: { in: ['PENDING', 'IN_APPROVAL']}},
        orderBy: { createdAt: 'desc' }
      });
      if (existing) {
        await prisma.dataSnapshot.update({ where: { id: existing.snapshotAfterId }, data: { rawData: afterSnapshot.rawData, diffData: afterSnapshot.diffData } as any });
        await prisma.dataSyncRequest.update({ where: { id: existing.id }, data: { changedFields: changedFields as any } as any });
        continue;
      }
      // 4. 建新的
      const sb = await prisma.dataSnapshot.create({ data: beforeSnapshot as any });
      const sa = await prisma.dataSnapshot.create({ data: afterSnapshot as any });
      await prisma.dataSyncRequest.create({
        data: {
          surveyTableId, surveyRecordId,
          levyTableId: rf.levyTableId, levyRecordId: rec.id,
          levyRelationFieldId: rf.id,
          changedFields: changedFields as any,
          snapshotBeforeId: sb.id, snapshotAfterId: sa.id,
          status: 'PENDING', approvalCategory: 'DATA_SYNC',
          createdBy: operatorId,
        } as any
      });
    }
  }
}

/** 返回 before/after snapshot draft + changedFields (keyed by surveyFieldId) */
async function buildSyncSnapshots(levyTableId: string, levyRec: any, fieldMapping: any[], newSurveyData: any, levyRelFieldId: string) {
  const before = { ...(levyRec.data || {}) };
  const after  = { ...before };
  const changedFields: any = {};

  // 取到调查记录再按 mapping 算（传进来的 newSurveyData 是 DataRecord.data 的 JSON 对象，key=fieldId 还是 field.name？需要和现有 dynamic-form 存的一致；实际 value 就按 newSurveyData[surveyFieldId] 取，不一致再调整）
  for (const m of fieldMapping) {
    const oldVal = before[m.levyFieldId];
    const newVal = newSurveyData?.[m.surveyFieldId];
    if (JSON.stringify(oldVal) !== JSON.stringify(newVal)) {
      after[m.levyFieldId] = newVal;
      changedFields[m.surveyFieldId] = { old: oldVal, new: newVal, mappedLevyFieldId: m.levyFieldId };
    }
  }

  return {
    beforeSnapshot: { tableId: levyTableId, tableName: '', recordId: levyRec.id, module: 'LEVY', snapshotType: 'SYNC_BEFORE', rawData: before as any, diffData: null, relatedSyncId: 'TBD' },
    afterSnapshot:  { tableId: levyTableId, tableName: '', recordId: levyRec.id, module: 'LEVY', snapshotType: 'SYNC_AFTER',  rawData: after  as any, diffData: { old: before, new: after, changedFields } as any, relatedSyncId: 'TBD' },
    changedFields,
  };
}
```

```ts
// web/app/api/levy-relation/survey-records/route.ts
import { NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';

export const dynamic = 'force-dynamic';

export async function GET(req: Request) {
  try {
    const { searchParams } = new URL(req.url);
    const surveyTableId = searchParams.get('surveyTableId');
    const displayFieldId = searchParams.get('displayFieldId');
    const q = (searchParams.get('q') || '').trim();
    const onlyApproved = searchParams.get('onlyApproved') !== 'false';
    const page = Math.max(1, parseInt(searchParams.get('page') || '1'));
    const pageSize = Math.min(50, parseInt(searchParams.get('pageSize') || '20'));

    if (!surveyTableId) return NextResponse.json({ code: 400, message: 'surveyTableId 必须' }, { status: 400 });

    // 先查 surveyTable 的所有字段（供 data JSON key 引用）
    const surveyFields = await prisma.tableField.findMany({ where: { tableId: surveyTableId }});
    const displayField = surveyFields.find(f => f.id === displayFieldId) || surveyFields.find(f => f.fieldType === 'TEXT');
    if (!displayField) return NextResponse.json({ code: 400, message: 'displayFieldId 无效' }, { status: 400 });

    // 过滤：如果 onlyApproved，要求 data.approvalStatus = 'APPROVED'（按现有调查审批的字段约定）
    const where: any = { tableId: surveyTableId, deletedAt: null };
    if (onlyApproved) {
      const apprField = surveyFields.find(f => f.fieldType === 'APPROVAL_STATUS');
      if (apprField) where[`data->>'${apprField.id}'`] = 'APPROVED';
    }
    // 关键字：在 displayField 里 ilike
    if (q) {
      const key = `data->>'${displayField.id}'`;
      where[key] = { contains: q, mode: 'insensitive' };
    }
    // NOTE: Prisma JsonFilter 用 path: [displayField.id], string_contains: { value: q, mode: 'insensitive' } 写法更稳，这里按实现修正
    const prismaWhere: any = {
      tableId: surveyTableId, deletedAt: null,
      data: onlyApproved ? apprField ? { path: [apprField.id], equals: 'APPROVED' } : undefined : undefined,
    };
    if (q && displayField) prismaWhere.data = { ...prismaWhere.data, path: [displayField.id], string_contains: { value: q, mode: 'insensitive' } };

    const [total, rows] = await Promise.all([
      prisma.dataRecord.count({ where: prismaWhere }),
      prisma.dataRecord.findMany({ where: prismaWhere, skip: (page - 1) * pageSize, take: pageSize, orderBy: { createdAt: 'desc' } }),
    ]);

    const items = rows.map(r => ({
      recordId: r.id,
      display: String((r.data as any)?.[displayField.id] ?? '(空)'),
      mapped: Object.fromEntries(surveyFields.map(f => [f.id, (r.data as any)?.[f.id]])),
    }));
    return NextResponse.json({ code: 0, total, page, pageSize, items });
  } catch (e: any) {
    return NextResponse.json({ code: 500, message: e?.message || String(e) }, { status: 500 });
  }
}
```
