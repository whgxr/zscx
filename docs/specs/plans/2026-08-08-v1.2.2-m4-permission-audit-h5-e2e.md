```ts
// web/lib/permission-utils.ts
import type { PermissionTreeState } from '@prisma/client'; // 实际类型用 Prisma Json 定义

export interface PermissionTreeNode {
  key: string;
  label: string;
  children?: PermissionTreeNode[];
  ops?: { key: string; label: string; }[]; // 叶子的操作列表
}
export type PermissionGrantState = 'allow' | 'deny' | 'unset';

/** 把一个角色已有 permissionTree（M1 schema）+ 请求的 path → 判定 */
export function resolvePermission(tree: any, path: string[]): PermissionGrantState {
  let node = tree;
  for (const seg of path) {
    if (!node) break;
    if (node.children?.[seg]) node = node.children[seg];
    else if (node.tables?.[seg]) node = node.tables[seg];
    else if (node.operations?.[seg]) node = node.operations[seg];
    else node = undefined;
  }
  if (!node) return 'unset';
  if (node.state === 'allow') return 'allow';
  if (node.state === 'deny') return 'deny';
  // 叶子若未设，看模块级；模块若 allow 则 allow（继承式）
  return 'unset';
}
/** 三态 checkbox：一个节点下若全 allow → checked=true；全 deny→partial；混合→partial；根未设→unchecked */
export function computeTriState(node: any): boolean | 'indeterminate' {
  if (!node.children && !node.tables && !node.operations) return node.state === 'allow';
  const all = Object.values({ ...(node.children ?? {}), ...(node.tables ?? {}), ...(node.operations ?? {}) });
  if (!all.length) return node.state === 'allow';
  const states = all.map(computeTriState as any);
  const a = states.filter(s => s === true).length,
        i = states.filter(s => s === 'indeterminate').length,
        d = states.filter(s => s === false).length;
  if (a === states.length) return true;
  if (d === states.length) return false;
  return 'indeterminate';
}
```

```ts
// 在每个 route 顶部用：
export async function requirePermission(path: string[]) {
  const me = await currentUser();
  if (!me) throw new Error('未登录');
  const isAdmin = await isUserAdmin(me.id);
  if (isAdmin) return true;
  // 读当前用户所有 roles
  const roles = await prisma.userRole.findMany({ where: { userId: me.id }, include: { role: true }});
  const combinedAllow = roles.some(ur => resolvePermission(ur.role.permissionTree, path) === 'allow');
  const anyDeny = roles.some(ur => resolvePermission(ur.role.permissionTree, path) === 'deny');
  if (anyDeny || !combinedAllow) {
    const err: any = new Error('权限不足（缺少 ' + path.join(' > ') + '）');
    err.code = 403;
    throw err;
  }
  return true;
}
```

```ts
export async function getPermissionDictionary() {
  // 1) SURVEY / LEVY 两大模块
  const cats = await prisma.tableCategory.findMany({ where: { module: { in: ['SURVEY','LEVY','BOTH']}}, include:{ tables: { include: { fields: false }}}});
  const tree: PermissionTreeNode[] = [
    { key: 'SURVEY', label: '调查模块', children: [] },
    { key: 'LEVY',   label: '征收模块', children: [] },
  ];
  for (const c of cats) {
    const mod = c.module === 'BOTH' ? ['SURVEY','LEVY'] : [c.module];
    for (const m of mod) {
      const cnode: PermissionTreeNode = { key: `CAT_${c.id}`, label: `分类：${c.name}`, children: [] };
      for (const t of c.tables) {
        const tnode: PermissionTreeNode = { key: t.id, label: `${t.displayName}(${t.name})`, ops: [
          { key:'canView',   label:'查看列表' },
          { key:'canDetail', label:'查看详情' },
          { key:'canCreate', label:'新增记录' },
          { key:'canEdit',   label:'编辑记录' },
          { key:'canDelete', label:'删除记录' },
          { key:'canApprove',label:'审批（发起&处理）' },
          { key:'canSync',   label:'发起/执行同步' },
          { key:'canPrint',  label:'生成&打印文书' },
        ]};
        cnode.children!.push(tnode);
      }
      tree.find(x=>x.key===m)!.children!.push(cnode);
    }
  }
  // 2) 系统模块：审批中心 / 审计中心 / 角色管理 / 模板管理 / 用户管理
  tree.push({ key:'SYSTEM', label:'系统管理', children: [
    { key:'APPROVAL_CENTER', label:'审批中心', ops:[{key:'canDesign',label:'设计工作流'},{key:'canProcess',label:'处理待办'}]},
    { key:'AUDIT', label:'审计日志中心', ops:[{key:'canViewAll',label:'查看全量'},{key:'canRollback',label:'强制回滚'},{key:'canExport',label:'导出'}]},
    { key:'ROLES', label:'角色管理', ops:[{key:'canView',label:'查看'},{key:'canEdit',label:'编辑'},{key:'canAssignTree',label:'编辑权限树'}]},

  ]});
```

```tsx
// permission-tree-editor.tsx
'use client';
import { Checkbox } from '@/components/ui/checkbox';
import { ChevronRight, ChevronDown } from 'lucide-react';
import { useState } from 'react';
import type { PermissionTreeNode } from '@/lib/permission-utils';

export function PermissionTreeEditor({ dict, initialTree, onSave }: any) {
  const [state, setState] = useState<any>(initialTree ?? {});  // 和 Role.permissionTree schema 对齐
  // 渲染：
  function Node({ node, pathKeys }: { node: PermissionTreeNode; pathKeys: string[] }) {
    const [open, setOpen] = useState(true);
    return (
      <div className="ml-2 border-l pl-3">
        <div className="flex items-center gap-2 py-0.5">
          {(node.children?.length ?? (node.ops?.length ?? 0) > 0) ? (
            open ? <ChevronDown size={14} onClick={()=>setOpen(false)}/> : <ChevronRight size={14} onClick={()=>setOpen(true)}/>
          ) : <span className="w-3.5"/>}
          <Checkbox />
          <span className="text-xs">{node.label}</span>
        </div>
        {open && node.children?.map(c => <Node key={c.key} node={c} pathKeys={[...pathKeys, c.key]}/>)}
        {open && node.ops?.map(op => (
          <div key={op.key} className="ml-5 flex items-center gap-2 py-0.5">
            <Checkbox /><span className="text-[11px] text-slate-600">{op.label}</span>
          </div>
        ))}
      </div>
    );
  }
  return (<div>
    <div className="mb-2 flex gap-2">
      <button className="text-xs rounded border px-2 py-1" onClick={()=>onSave(state)}>💾 保存此角色的权限树</button>
      <button className="text-xs rounded border px-2 py-1">✅ 全部勾选</button>
      <button className="text-xs rounded border px-2 py-1">❌ 全部取消</button>
    </div>
    {dict.map(r => <Node key={r.key} node={r} pathKeys={[r.key]}/>)}
  </div>);
}
```
