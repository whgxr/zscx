> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 重做 v1.2.1 「审批流程设计器」为**可视化拖拽画布**（工具箱/连线/条件分支/并行/抄送），重写审批引擎 Service（会签/或签/超时/加签/转办/强制回滚）；完成两种业务审批触发：**① 征收新增/修改（DATA_EDIT）提交≠生效，要审批通过才写库**；**② 调查→征收同步（DATA_SYNC）通过后才执行同步**；H5/Dashboard 两端通用的「审批详情 + 时间轴 + 差异回放 + 处理面板」组件。

**Architecture:** 保留现有审批 DB 表的所有原字段 + M1 扩的 V2 字段；老流程实例（nodeIds 数组）继续走老判定，新 ACTIVE 流程（nodeConfig JSON 画布版）走新引擎 `ApprovalEngineV2`；前端设计器用 `@xyflow/react`（原 ReactFlow，MIT 免费，拖拽连线成熟）或 shadcn 自带简单 dnd（优先推荐 xyflow 减少造轮子）；后端审批引擎是纯函数 service（便于单元测试），通过后回调钩子写 DataRecord / 执行同步 / 发 Notification。

**Tech Stack:** React 18 + Next.js App Router (Route Handlers) + Prisma PostgreSQL；前端新依赖 `@xyflow/react`（≈1.2MB，npm），`dnd-kit/core`（已有 shadcn 可复用的话不加）；后端 0 新依赖。

- [ ] **Step 2: 实现 `web/lib/approval-engine-v2.ts` 纯函数状态机（不直接写 DB，读 nodeConfig → 返回 runtime 结构，DB 写由 Route Handler 调）**

```ts
// web/lib/approval-engine-v2.ts
// 注意：这个文件是纯逻辑，不依赖 prisma；外部 DB 读写在 route.ts 里
export interface WorkflowNodeConfig {
  id: string;
  type: 'START' | 'APPROVAL' | 'CONDITION' | 'PARALLEL_BEGIN' | 'PARALLEL_END' | 'CC' | 'END';
  kind?: 'SINGLE' | 'COUNTERSIGN' | 'ORSIGN';
  approverSource?: { type: 'USER'|'ROLE'|'FORMULA'|'FORM_FIELD_LEADER'; value?: any };
  approverIds?: string[]; // FORMULA 算出来的用户 ID 数组，route 层传入
  ccUserIds?: string[];
  conditionExpr?: string;   // CONDITION：用 JS 表达式，route 层用 new Function() 算好 true/false 传进 runtime
}
export interface WorkflowDiagram { nodes: WorkflowNodeConfig[]; edges: Array<{ id?:string; source:string; target:string }>; }

export type Phase = 'PENDING' | 'ACTIVE' | 'DONE' | 'SKIPPED';
export interface NodeRuntimeState {
  phase: Phase;
  approvals?: Array<{ actorId: string; pass: boolean; reason?: string; at: number }>;  // 每一位处理人
  pendingApproverIds?: string[]; // 本节点等待处理的人（或签/会签都用）
  active?: boolean;
  skippedReason?: string;
  ccSentTo?: string[];
}
export interface Runtime {
  status: 'RUNNING' | 'COMPLETED' | 'REJECTED';
  nodeStates: Record<string, NodeRuntimeState>;
  rejectReason?: string;
  rollback?: boolean;
}

export function buildInitialRuntime(diagram: WorkflowDiagram, overrides?: Partial<Runtime>): Runtime {
  const nodeStates: Record<string, NodeRuntimeState> = {};
  for (const n of diagram.nodes) {
    nodeStates[n.id] = { phase: n.type === 'START' ? 'DONE' : 'PENDING' };
    if (n.type === 'APPROVAL') {
      nodeStates[n.id].pendingApproverIds = n.approverIds ?? [];
      nodeStates[n.id].approvals = [];
    }
  }
  const rt: Runtime = { status: 'RUNNING', nodeStates, ...overrides };
  // 激活起点后继
  activateNext(rt, diagram, getNode(diagram, 'START'));
  return rt;
}

export function getNode(d: WorkflowDiagram, id: string) { return d.nodes.find(n => n.id === id)!; }
export function nextNodes(d: WorkflowDiagram, fromId: string) {
  return d.edges.filter(e => e.source === fromId).map(e => getNode(d, e.target));
}
function activateNext(rt: Runtime, d: WorkflowDiagram, from: WorkflowNodeConfig) {
  if (from.type === 'END') rt.status = 'COMPLETED';
  for (const t of nextNodes(d, from.id)) {
    // 若节点前一个都没 DONE，别提前激活（PARALLEL 汇合：后面 PARALLEL_END 处理）
    if (t.type === 'PARALLEL_END') {
      const prevs = d.edges.filter(e => e.target === t.id).map(e => e.source);
      if (!prevs.every(p => rt.nodeStates[p]?.phase === 'DONE')) continue;
      rt.nodeStates[t.id].phase = 'DONE';
      activateNext(rt, d, t);
      continue;
    }
    if (t.type === 'CC') {
      rt.nodeStates[t.id] = { phase: 'DONE', ccSentTo: t.ccUserIds };
      activateNext(rt, d, t);
      continue;
    }
    if (t.type === 'APPROVAL') {
      if (t.kind === 'COUNTERSIGN' || t.kind === 'ORSIGN') {
        rt.nodeStates[t.id] = { phase: 'ACTIVE', pendingApproverIds: t.approverIds ?? [], approvals: [] };
      } else {
        rt.nodeStates[t.id] = { phase: 'ACTIVE', pendingApproverIds: t.approverIds ?? [], approvals: [] };
      }
    }
    if (t.type === 'END') {
      rt.nodeStates[t.id] = { phase: 'DONE' };
      rt.status = 'COMPLETED';
    }
  }
}

export interface NodeEvent { nodeId: string; event: 'PASS'|'REJECT'; actorId: string; reason?: string; }
export function applyNodeEvent(rt: Runtime, ev: NodeEvent): Runtime {
  const st = rt.nodeStates[ev.nodeId];
  if (!st || st.phase !== 'ACTIVE') throw new Error(`节点 ${ev.nodeId} 不在 ACTIVE，当前 ${st?.phase}`);
  st.approvals = st.approvals ?? [];
  if (st.approvals.find(a => a.actorId === ev.actorId)) throw new Error(`处理人 ${ev.actorId} 已操作`);
  if (!(st.pendingApproverIds ?? []).includes(ev.actorId)) throw new Error(`${ev.actorId} 不是本节点审批人`);

  if (ev.event === 'REJECT') {
    st.approvals.push({ actorId: ev.actorId, pass:false, reason: ev.reason, at: Date.now() });
    st.phase = 'DONE';
    rt.status = 'REJECTED';
    rt.rejectReason = ev.reason;
    return rt;
  }
  // PASS
  st.approvals.push({ actorId: ev.actorId, pass:true, reason: ev.reason, at: Date.now() });
  const nodeKind = (st as any)._nodeKind ?? 'SINGLE';
  // orsign：1 人过即过；countersign：pendingApproverIds 所有人都 pass
  const allPassed = st.pendingApproverIds!.every(uid => st.approvals!.find(a => a.actorId === uid && a.pass));
  if (nodeKind === 'ORSIGN' ? true : allPassed) {
    // 注意 orsign 只要有人 pass 就直接 DONE（其余人可不再处理；若要严谨把或签/会签的通过条件拆分如下）
    const done = (nodeKind === 'COUNTERSIGN') ? allPassed : true;
    if (done) {
      st.phase = 'DONE';
      // 找到源 node 用的 nodeId 对应配置（Runtime 没存，需要外部在 route 层把 diagram 传进来；为纯函数解耦我们新增 rebuildActivate）
      (st as any)._needActivateFrom = ev.nodeId;
    }
  }
  return rt;
}

/** 外部调用：事件处理完后，用此函数激活下一节点（必须把 diagram 传进来）*/
export function progressAfterNodePass(rt: Runtime, d: WorkflowDiagram, passedNodeId: string): Runtime {
  if (rt.nodeStates[passedNodeId].phase === 'DONE') {
    activateNext(rt, d, getNode(d, passedNodeId));
  }
  return rt;
}

/** 条件节点分支：route 层算出 conditionResult=true/false 后调用，只激活目标分支 */
export function branchCondition(rt: Runtime, d: WorkflowDiagram, conditionNodeId: string, result: boolean): Runtime {
  const cond = getNode(d, conditionNodeId);
  rt.nodeStates[conditionNodeId].phase = 'DONE';
  const out = d.edges.filter(e => e.source === conditionNodeId);
  const target = out[result ? 0 : 1]; // 约定：edges[0]=true 分支, edges[1]=false 分支
  if (!target) throw new Error('CONDITION 节点需要 2 条出边');
  activateNext(rt, d, getNode(d, target.target));
  return rt;
}
```

import { NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { requireRole } from '@/lib/auth';

export async function GET(_req: Request, { params }: { params: { id: string }}) {
  const wf = await prisma.approvalWorkflow.findUnique({ where: { id: params.id }});
  if (!wf) return NextResponse.json({ code: 404 }, { status: 404 });
  return NextResponse.json({ code: 0, nodeConfig: wf.nodeConfig, statusV2: wf.statusV2, version: wf.version });
}

export async function PUT(req: Request, { params }: { params: { id: string }}) {
  await requireRole(['ADMIN', 'MANAGER']);
  const { nodeConfig } = await req.json();
  if (!nodeConfig?.nodes?.length) return NextResponse.json({ code:400, message:'nodeConfig 无效' }, { status:400 });
  const saved = await prisma.approvalWorkflow.update({
    where: { id: params.id },
    data: { nodeConfig: nodeConfig as any, statusV2: 'DRAFT' },
  });
  return NextResponse.json({ code: 0, statusV2: saved.statusV2 });
}

// web/app/api/approval/workflows/[id]/publish/route.ts
import { NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { requireRole } from '@/lib/auth';
export async function POST(_req: Request, { params }: { params: { id: string }}) {
  await requireRole(['ADMIN']);
  const wf = await prisma.approvalWorkflow.findUnique({ where: { id: params.id }});
  if (!wf?.nodeConfig) return NextResponse.json({ code:400, message:'nodeConfig 未保存' }, { status:400 });
  await prisma.$transaction([
    // 同一张 tableId 的其他 ACTIVE workflow 置 DEPRECATED
    prisma.approvalWorkflow.updateMany({
      where: { tableId: wf.tableId, statusV2: 'ACTIVE', id: { not: wf.id }},
      data: { statusV2: 'DEPRECATED' },
    }),
    prisma.approvalWorkflow.update({
      where: { id: wf.id },
      data: { statusV2: 'ACTIVE', version: { increment: 1 }, publishedAt: new Date() /* publishedBy 从 session 取 */ },
    }),
  ]);
  return NextResponse.json({ code: 0 });
}

// web/app/api/approval/instances/[id]/approve/route.ts
import { NextResponse } from 'next/server';
import { prisma } from '@/lib/prisma';
import { requireLogin, currentUser } from '@/lib/auth';
import { buildInitialRuntime, applyNodeEvent, progressAfterNodePass } from '@/lib/approval-engine-v2';

export async function POST(req: Request, { params }: { params: { id: string }}) {
  const me = await currentUser();
  if (!me) return new Response('未登录', { status: 401 });
  const body = await req.json();
  const { nodeInstanceId, opinion } = body;
  // 1. 查 instance + 所属 workflow(nodeConfig)
  const inst = await prisma.approvalInstance.findUnique({
    where: { id: params.id },
    include: { workflow: true, nodeInstances: true },
  });
  if (!inst) return NextResponse.json({ code: 404 }, { status: 404 });
  if (!inst.workflow.nodeConfig) return NextResponse.json({ code: 400, message:'老流程请走 legacy engine' }, { status: 400 });

  // 2. 确定当前 ACTIVE 节点 & 我是不是处理人（这里简化按 pendingApproverIds；加签/转办的 pendingApproverIds 在 add-signer/transfer route 动态扩进当前 nodeInstance 的 pending 字段存到 ApprovalNodeInstance.config）
  const curNode = inst.nodeInstances.find(n => n.id === nodeInstanceId);
  if (!curNode || curNode.status !== 'PENDING') return NextResponse.json({ code:400, message:'节点非待处理' }, { status:400 });

  // 3. 把 ApprovalNodeInstance.config 中记录的 "当前审批人ids"（含加签/转办）拼进 runtime
  const wfDiagram = inst.workflow.nodeConfig as any;
  wfDiagram.nodes = wfDiagram.nodes.map((n: any) => n.id === curNode.workflowNodeId
    ? { ...n, approverIds: (curNode.config as any)?.pendingApproverIds ?? n.approverIds }
    : n);
  // 4. 用 DB 已有的通过/驳回历史（所有已存 nodeInstances 结果）rebuild runtime 状态
  let rt = rebuildRuntimeFromDb(wfDiagram, inst.nodeInstances, curNode.workflowNodeId);
  rt = applyNodeEvent(rt, { nodeId: curNode.workflowNodeId, event:'PASS', actorId: me.id, reason: opinion });
  rt = progressAfterNodePass(rt, wfDiagram, curNode.workflowNodeId);

  // 5. 写 DB：当前 nodeInstance → APPROVED；若整体 COMPLETED，instance.status → APPROVED + 回调执行后续动作
  const finalStatus = rt.status === 'COMPLETED' ? 'APPROVED' : (rt.status === 'REJECTED' ? 'REJECTED' : inst.status);
  await prisma.$transaction(async tx => {
    await tx.approvalNodeInstance.update({ where: { id: curNode.id },
      data: { status: 'APPROVED', result: opinion ?? '', approverId: me.id, handledAt: new Date() }});
    // 其他节点仍 PENDING 的按 rt.nodeStates 写一批（简略）
    if (rt.status === 'COMPLETED' || rt.status === 'REJECTED') {
      await tx.approvalInstance.update({ where: { id: inst.id }, data: { status: finalStatus, approvedAt: new Date(), approvedBy: me.id }});
    }
    // —— v2 通过后钩子：按 instance.category 做不同事 ——
    if (finalStatus === 'APPROVED') {
      await executePostApprovalActions(tx, inst); // 下方定义
    }
  });
  return NextResponse.json({ code: 0, newStatus: finalStatus });
}

// ===== helper =====
async function executePostApprovalActions(tx: any, inst: any) {
  // (a) DATA_EDIT：把 EDIT_AFTER 快照写进 DataRecord（若原来是编辑）
  if (inst.category === 'DATA_EDIT') {
    const [beforeId, afterId] = inst.snapshotIds || [];
    if (!afterId) return;
    const snapAfter = await tx.dataSnapshot.findUnique({ where: { id: afterId }});
    if (!snapAfter) return;
    await tx.dataRecord.upsert({
      where: { id: snapAfter.recordId },
      create: { id: snapAfter.recordId, tableId: snapAfter.tableId, data: snapAfter.rawData as any, version: 1 },
      update: { data: snapAfter.rawData as any, updatedAt: new Date() },
    });
    await tx.operationLog.create({ data: { /* user, table, record, diffData 等 */ } as any });
  }
  // (b) DATA_SYNC：调用 M1 的同步 execute 逻辑（直接调用 sync-requests/execute route 相同的内部函数）
  if (inst.category === 'DATA_SYNC' && inst.syncRequestId) {
    // executeSync 见 Task6 的内部导出
    const { executeApprovedSync } = await import('@/lib/levy-sync-executor');
    await executeApprovedSync(tx, inst.syncRequestId, (inst.approvedBy || inst.createdBy)!);
  }
  // (c) 发 Notification（站内）
  await tx.notification.createMany({
    data: [{ userId: inst.submitterId, title: `审批通过`, content: `您的审批单已通过`, type: 'APPROVAL' }]
  });
}
function rebuildRuntimeFromDb(_d: any, _nodes: any[], _active: string) {
  return buildInitialRuntime(_d); // 简化：真实实现按 nodeInstances 把已处理节点 phase = DONE / REJECTED
}

```ts
import type { Prisma } from '@prisma/client';
export async function executeApprovedSync(tx: Prisma.TransactionClient, syncRequestId: string, approvedByUserId: string) {
  const req = await tx.dataSyncRequest.findUnique({ where: { id: syncRequestId }});
  if (!req || req.status !== 'APPROVED') return;
  const snapAfter = await tx.dataSnapshot.findUnique({ where: { id: req.snapshotAfterId! }});
  if (!snapAfter) return;
  // merge newVal into existing DataRecord：只改 changedFields 涉及的字段
  const current = await tx.dataRecord.findUnique({ where: { id: req.levyRecordId }});
  const newData = { ...(current?.data as any || {}) };
  for (const [sFid, chg] of Object.entries(req.changedFields as any)) {
    const levyFid = (chg as any).mappedLevyFieldId;
    newData[levyFid] = (chg as any).new;
  }
  await tx.dataRecord.update({ where: { id: req.levyRecordId }, data: { data: newData as any }});
  // 再写 SYNC_AFTER 快照
  await tx.dataSnapshot.create({ data: {
    tableId: req.levyTableId, tableName:'', recordId: req.levyRecordId, module:'LEVY',
    snapshotType:'SYNC_AFTER', rawData: newData as any, relatedSyncId: req.id,
  } as any });
  await tx.dataSyncRequest.update({ where: { id: req.id }, data: { status: 'SYNCED', executedAt: new Date(), approvedBy: approvedByUserId }});
  await tx.operationLog.create({ data: { /* action=SYNC_EXECUTED */ } as any });
}
```
